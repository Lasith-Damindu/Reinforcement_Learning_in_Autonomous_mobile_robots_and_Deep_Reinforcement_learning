for ep in range(episodes):
    eps = max(eps_min, eps0 * (1 - ep / (0.6 * episodes)))    s = env.reset(); done, tot = False, 0    while not done:
        a = rng.integers(4) if rng.random() < eps else int(np.argmax(Q[s]))        s2, r, done, ok = env.step(a)        target = r + (0 if (done and env.t < env.max_steps) else gamma * Q[s2].max())        Q[s, a] += alpha * (target - Q[s, a]); s, tot = s2, tot + r    rewards.append(tot); succ.append(ok)
